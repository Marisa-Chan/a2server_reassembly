Packet::field_0x4 -> order_mode -- set by packet dispatcher from id: 0=single-unit op, 1/2=group order (no/xy params), 3=group order with unit/building target
Packet::field_0x5 -> from_player_id -- sender player id (mirrors to_player_id); NOT player_id to avoid shadowing PacketJoin::player_id
PacketInfo::field_0xa/field_a1..a4 -> param1/param1_b0..b3 -- generic first dword parameter; bytes used for body/reaction/mind/spirit in the char-gen packet (0x48)
PacketInfo::field_0xe/field_e1..e4 -> param2/param2_b0..b3 -- generic second dword parameter; bytes carry main_sphere/face/color in the char-gen packet (0x48)
PacketJoin::field_0xc -> color -- player color sent in join-notify (0x96), read into MapPlayerData(index, color); overlaid by score dword for chat packets
PacketItemOperation::field_0xc -> y -- low byte of the target tile y in group orders (paired with y_high); also renamed already-named unit_id (+0xA) to x for coherence: it holds the target x, server reads it as uint8_t x
PacketItemOperation::field_0xd -> y_high -- high byte of the target tile y (writers store y>>8 here)
PacketItemOperation::field_0xe -> param -- word parameter, meaning depends on op id: target unit/building id (0x19/1B/1E/24/25), bag slot (0x21)
PacketItemOperation::field_0x10 -> slot -- spell book / inventory slot index (low byte; BOOK_POS_TO_SPELL_ID or inventory position)
PacketItemOperation::field_0x11 -> slot_high -- high byte of the slot index (writers store slot>>8 here)
PacketCmd::field_0xe -> param1 -- first generic word parameter of the 0x22 command packet (inventory pos / slot / drop coords)
PacketCmd::field_0x10 -> param2 -- second generic word parameter (target slot / packed drop x|y<<8)
PacketCmd::field_0x12 -> param3 -- third generic word parameter (init 1 in ctor; item count in inventory ops)
Packet3Dwords::field_0xa -> value1 -- first dword payload (player id in the Allods2 join-compat packet 0x02)
Packet3Dwords::field_0xe -> value2 -- second dword payload
Packet3Dwords::field_0x12 -> value3 -- third dword payload (player params word in join-compat 0x02)
Packet3Dwords::field_0x16 -> text -- 32-byte char payload, player name in join-compat 0x02
PacketAbility::field_0xa -> unit_id -- attacking unit id (word, from unit->building_id)
PacketAbility::field_0xc -> angle -- attack direction angle (client stores to unit->action_dir)
PacketAbility::field_0xd -> segments -- attack animation segments/ticks (charge+relax; client stores to unit->action_segments)
PacketMoveCmd::field_0xa -> unit_id -- caster/effect owner unit id (word)
PacketMoveCmd::field_0xc -> type_id -- projectile/effect type (g_ProjectileInfos index, spell_id*2+8)
PacketMoveCmd::field_0xd -> x -- target tile x (low byte of target unit id for homing projectiles)
PacketMoveCmd::field_0xe -> y -- target tile y (high byte of target unit id for homing projectiles)
PacketMoveCmd::field_0xf -> segments -- projectile animation segments / cast delay (client stores to pj->action_segments)
PacketEffect::field_0xa -> unit_id -- unit the effect plays on
PacketAoeZone::field_0xa -> type_id -- AoE effect type id
PacketAoeZone::field_0xf -> set_flag -- nonzero = mark zone cells as affected, 0 = unmark them
PacketMount::unit_id (+0xC) -> target_id -- holds the mount/ride TARGET unit id (writer sets cast_target->building_id); freed the unit_id name for +0xA
PacketMount::field_0xa -> unit_id -- acting (riding) unit id
PacketUnitStateVec::field_0xc -> mode -- update kind: 1=equipment TLV entries, 2=inventory range; bit 0x80 marks a partial inventory range

NetStru2::field_0x2a8 -> is_claimed -- connection already bound to a player/login (guard for 0xD0 reconnect claim, set on HAT login and non-DPlay connect)
A2NetSock::field_0x4c -> next_out_pktid -- packet id assigned to the next outgoing non-guaranteed packet

GO_11c::field_0x0/2/4 -> x/y/z, field_0x6 -> vfx_type, field_0x7 -> phase -- transient VFX element: screen-space offsets, g_ProjectileInfos sprite id, anim phase (ctor params renamed to match)
